let audioEnabled = true;
let audioCtx;
function initAudio(){
  if(!audioCtx){ audioCtx = new(window.AudioContext || window.webkitAudioContext)(); }
}
function playBeep(freq=800,type="sine",duration=.05){
  if(!audioEnabled) return;
  try{
    initAudio();
    const osc=audioCtx.createOscillator();
    const gain=audioCtx.createGain();
    osc.type=type; osc.frequency.value=freq;
    gain.gain.setValueAtTime(.035,audioCtx.currentTime);
    gain.gain.exponentialRampToValueAtTime(.001,audioCtx.currentTime+duration);
    osc.connect(gain); gain.connect(audioCtx.destination);
    osc.start(); osc.stop(audioCtx.currentTime+duration);
  }catch(e){}
}

const typingPhrases=["INITIALIZING SECURE ARGOS CHANNEL...", "AUTHENTICATING USER CLEARANCE...", "EPISODE 01: BAYANG PERTAMA LOADED...", "AWAITING COMMAND INPUT..."];
function startTyping(){
  const target=document.getElementById("typingText");
  let phrase=0, index=0, deleting=false;
  function type(){
    const current=typingPhrases[phrase];
    if(deleting){ index--; target.textContent=current.substring(0,index); }
    else{
      index++; target.textContent=current.substring(0,index);
      if(index%3===0) playBeep(1100,"sine",.008);
    }
    let speed=deleting?25:45;
    if(!deleting && index===current.length){ speed=1800; deleting=true; }
    else if(deleting && index===0){ deleting=false; phrase=(phrase+1)%typingPhrases.length; speed=400; }
    setTimeout(type,speed);
  }
  type();
}

    <article class="ep-card ${e.hasScript?"highlight":""}" onclick="openEpisode('${e.id}')">
      <div class="ep-num">${e.num} ${e.hasScript?" • FULL SCRIPT":""}</div>
      <div class="ep-title">${e.title}</div>
      <div class="ep-desc">${e.desc}</div>
      <button class="btn ${e.hasScript?"btn-primary":""}">${e.hasScript ? "▶ READ FULL SCRIPT" : "🔒 CLASSIFIED"}</button>
    </article>
  `).join("");
}

function openModal(modalId) { initAudio(); playBeep(900, "square", 0.05); document.getElementById(modalId).classList.add("active"); }
function closeModal(modalId) { playBeep(400, "sine", 0.05); document.getElementById(modalId).classList.remove("active"); }
document.querySelectorAll('.close-modal').forEach(btn => {
  btn.addEventListener('click', (e) => { const modalId = e.currentTarget.getAttribute('data-close'); if (modalId) closeModal(modalId); });
});
document.querySelectorAll('.modal').forEach(modal => {
  modal.addEventListener('click', (e) => { if (e.target === modal) { modal.classList.remove('active'); playBeep(400, "sine", 0.05); } });
});

function openDossier(id) {
  const char = CHARACTERS.find(c => c.id === id); if (!char) return;
  const content = document.getElementById("dossierContent");
  content.innerHTML = `
    <div style="display:flex; gap:20px; margin-bottom:20px; align-items:center;">
      <div style="font-size:3.5rem; filter:grayscale(0.2); background:var(--panel); padding:15px; border-radius:15px; border:1px solid var(--border);">${char.avatar}</div>
      <div>
        <h2 style="color:var(--cyan); font-size:1.4rem; margin-bottom:5px;">${char.name}</h2>
        <div style="font-size:0.75rem; color:var(--muted); margin-bottom:4px; letter-spacing:1px;">ROLE // ${char.role}</div>
        <div style="font-size:0.75rem; color:var(--muted); margin-bottom:4px; letter-spacing:1px;">STATUS // ${char.status}</div>
        <div style="font-size:0.75rem; color:var(--red); font-weight:bold; letter-spacing:1px;">CLEARANCE // ${char.clearance}</div>
      </div>
    </div>
    <p style="font-size:0.85rem; line-height:1.7; color:#e0e5ea;">${char.bio}</p>
  `;
  openModal("dossierModal");
}

function openEpisode(id) {
  const ep = EPISODES.find(e => e.id === id); if (!ep) return;
  const content = document.getElementById("episodeContent");
  document.getElementById("epClassTitle").textContent = ep.hasScript ? "DECRYPTED SCRIPT" : "CLASSIFIED BRIEFING";
  let html = `<h2 style="color:var(--green); font-size:1.3rem; margin-bottom:10px;">${ep.num}: ${ep.title}</h2><p style="font-size:0.8rem; color:var(--muted); line-height:1.6; margin-bottom:20px;">${ep.desc}</p>`;
  if (ep.hasScript && id === "ep1") html += `<div class="script-container">${EPISODE_1_TEXT}</div>`;
  else if (ep.hasScript) html += `<div class="script-container">DATA CORRUPTED OR UNAVAILABLE.</div>`;
  else html += `<div class="locked-panel" style="margin-top:20px; border:1px solid rgba(255,59,99,.2); border-radius:12px; background:rgba(255,59,99,.05);"><div class="lock-icon" style="margin:15px auto;">🔒</div><h3 style="color:var(--red); font-family:var(--title); font-size:1rem; letter-spacing:2px;">ACCESS RESTRICTED</h3><p style="font-size:0.75rem; color:var(--muted); margin-top:8px;">FULL SCRIPT REQUIRES HIGHER CLEARANCE LEVEL.</p></div>`;
  content.innerHTML = html; openModal("episodeModal");
}

   SEARCH LOGIC
===================================================== */
document.getElementById("searchInput").addEventListener("input", e => {
  const query = e.target.value.toLowerCase().trim(); const results = document.getElementById("searchResults");
  if (!query) { results.innerHTML = ""; return; }
  const chars = CHARACTERS.filter(c => c.name.toLowerCase().includes(query) || c.role.toLowerCase().includes(query) || c.status.toLowerCase().includes(query));
  const eps = EPISODES.filter(ep => ep.title.toLowerCase().includes(query) || ep.desc.toLowerCase().includes(query));
  let html = "";
  if (chars.length > 0) {
    html += `<div style="font-size:.6rem;color:var(--cyan);margin:6px 0;letter-spacing:1px;">OPERATIVES / CHARACTERS</div>`;
    chars.forEach(c => { html += `<div style="padding:12px;border:1px solid var(--border);border-radius:10px;background:rgba(255,255,255,.02);cursor:pointer;margin-bottom:8px;transition:0.2s;" onclick="openDossier('${c.id}')" onmouseover="this.style.background='rgba(53,232,255,.05)'" onmouseout="this.style.background='rgba(255,255,255,.02)'"><b style="font-size:0.8rem;">${c.name}</b> <br><span style="color:var(--muted);font-size:.65rem;margin-top:4px;display:inline-block;">// ${c.role}</span></div>`; });
  }
  if (eps.length > 0) {
    html += `<div style="font-size:.6rem;color:var(--green);margin:6px 0;letter-spacing:1px;">EPISODE DATABASE</div>`;
    eps.forEach(ep => { html += `<div style="padding:12px;border:1px solid var(--border);border-radius:10px;background:rgba(255,255,255,.02);cursor:pointer;margin-bottom:8px;transition:0.2s;" onclick="openEpisode('${ep.id}')" onmouseover="this.style.background='rgba(57,247,164,.05)'" onmouseout="this.style.background='rgba(255,255,255,.02)'"><b style="font-size:0.8rem;color:var(--green);">${ep.num}</b> <span style="font-size:0.8rem;">— ${ep.title}</span></div>`; });
  }
  if (chars.length === 0 && eps.length === 0) html = `<div style="color:var(--muted);font-size:.7rem;padding:15px;text-align:center;border:1px dashed var(--border);border-radius:10px;">TIADA REKOD DIJUMPAI DALAM PANGKALAN DATA.</div>`;
  results.innerHTML = html;
});
